### 6.2 useCompactLayout.ts 增强

#### 6.2.1 增强 hasWailsRuntime() 检测

```typescript
function hasWailsRuntime(): boolean {
  if (typeof window === "undefined") return false;
  // 多种方式检测 Wails 环境
  const win = window as any;
  return Boolean(win.runtime) || Boolean(win.wails) || Boolean(win.go);
}
```

#### 6.2.2 修改 computeCompactLayout() 逻辑

当前逻辑（问题）：
```typescript
if (hasWailsRuntime()) {
  return false;  // Wails 环境一律返回非紧凑 → 如果检测失败就完蛋
}
```

改为：
```typescript
export function computeCompactLayout(width: number): boolean {
  // 第一层：Wails 桌面应用 —— 结合窗口宽度判断
  if (typeof window !== "undefined" && hasWailsRuntime()) {
    // Wails 环境：宽度大于紧凑阈值时使用桌面布局
    // 即使 Wails 中也允许小窗口时进入紧凑模式
    return width <= COMPACT_LAYOUT_MAX_WIDTH;
  }

  // 第二层：非移动 UA + 精细指针（桌面浏览器）
  if (!isMobileUserAgent() && hasFinePointerInput()) {
    return false;
  }

  // 第三层：宽度硬判断
  if (width >= DESKTOP_LAYOUT_MIN_WIDTH) return false;
  if (width <= COMPACT_LAYOUT_MAX_WIDTH) return true;

  // 灰色区间 (768 < width < 1024)
  if (isMobileUserAgent()) return true;
  if (hasFinePointerInput()) return false;
  return window.matchMedia("(pointer: coarse)").matches;
}
```

关键改变：
- Wails 检测到时不再一律返回 false，而是根据窗口宽度判断（≤768px 才紧凑）
- 增加了多种 Wails 检测方式（window.runtime / window.wails / window.go）
