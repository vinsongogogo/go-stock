### 4.1 LongTigerRank 主组件

```typescript
// frontend/src/app/components/LongTigerRank.tsx
import { useState, useEffect, useCallback } from 'react';
import { LongTigerRank as getLongTigerRank } from '../../../wailsjs/go/main/App';
import { KLineChart } from './KLineChart';
import { MoneyTrend } from './MoneyTrend';
import { Popover, PopoverContent, PopoverTrigger } from '../components/ui/popover';

export function LongTigerRank() {
  const [date, setDate] = useState<string>(formatDate(new Date()));
  const [data, setData] = useState<LongTigerRankData[]>([]);
  const [filteredData, setFilteredData] = useState<LongTigerRankData[]>([]);
  const [explanations, setExplanations] = useState<string[]>([]);
  const [selectedExplanation, setSelectedExplanation] = useState<string>('');
  const [loading, setLoading] = useState(false);

  // 获取数据（含自动回退逻辑）
  const fetchData = useCallback(async (currentDate: string, retryCount = 0) => {
    if (retryCount > 7) {
      setData([]);
      setFilteredData([]);
      return;
    }
    
    setLoading(true);
    try {
      const result = await getLongTigerRank(currentDate);
      if (result.length === 0) {
        // 递归查询前一日
        const prevDate = getPreviousDate(currentDate);
        fetchData(prevDate, retryCount + 1);
      } else {
        setData(result);
        setFilteredData(result);
        // 提取唯一的上榜原因
        const uniqueExplanations = [...new Set(result.map(item => item.EXPLANATION))];
        setExplanations(uniqueExplanations);
      }
    } catch (error) {
      console.error('获取龙虎榜数据失败:', error);
    } finally {
      setLoading(false);
    }
  }, []);

  // 筛选逻辑
  const handleFilter = (explanation: string) => {
    setSelectedExplanation(explanation);
    if (explanation) {
      setFilteredData(data.filter(item => item.EXPLANATION === explanation));
    } else {
      setFilteredData(data);
    }
  };

  return (
    <div className="space-y-4">
      {/* 筛选区域 */}
      <div className="flex gap-4">
        <DatePicker value={date} onChange={setDate} />
        <Select 
          value={selectedExplanation}
          onValueChange={handleFilter}
          options={explanations.map(e => ({ label: e, value: e }))}
          placeholder="上榜原因筛选"
        />
      </div>
      
      {/* 数据表格 */}
      <DataTable data={filteredData} loading={loading} />
    </div>
  );
}
```

## 六、风险与注意事项

### 6.1 技术风险

| 风险 | 影响 | 缓解措施 |
|-----|------|---------|
| Recharts 与 ECharts 功能差异 | K 线图效果不一致 | 使用 ComposedChart 组合实现，必要时引入 echarts-for-react |
| 数据量大导致性能问题 | 页面卡顿 | 实现数据分页或虚拟滚动 |
| Wails 绑定调用失败 | 数据无法加载 | 添加错误处理和重试机制 |

### 6.2 实现注意事项

1. **日期格式**：Vue 版本使用 `YYYY-MM-DD`，确保与后端 API 格式一致
2. **股票代码转换**：`SECUCODE` 格式为 `000001.SZ`，需要转换为 `sz000001` 调用 API
3. **颜色规范**：涨红跌绿，与现有组件保持一致
4. **暗色主题**：所有组件支持 `darkTheme` 属性
5. **加载状态**：数据请求时显示 loading 指示器
